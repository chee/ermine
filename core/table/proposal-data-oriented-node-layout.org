// names.ts: a children entry's names, sorted, never saved. made on first read; each fact one or
// two splices (binary search by code point, 0.4 µs at 100k); a whole-value fact drops it
const kept = new WeakMap<Entry<unknown>, string[]>()
export function sorted(children: Entry<unknown>): string[] {
	let s = kept.get(children)
	if (s === undefined) kept.set(children, s = Object.keys(children.value ?? {}).sort(byCodePoint))
	return s
}
export function onFact({ part, entry, patches }: Fact): void {
	const s = part === "children" && entry ? kept.get(entry) : undefined
	if (s === undefined) return
	for (const p of patches) {
		if (p.path.length === 0) return void kept.delete(entry!)
		if (p.move) drop(s, p.move.path[0] as string)
		if ("value" in p || p.move) add(s, p.path[0] as string) // add: nothing if there
		else drop(s, p.path[0] as string)
	}
}
// a window: one keys spot (any place here reruns it), then a slice
export function window(places: Reads, children: Entry<unknown>, id: Id, from: number, n: number) {
	places.readKeys("children", id)
	return sorted(children).slice(from, from + n)
}

- what a listing needs (is, size) lives in attributes, so a listing never loads a value: hot and

** Rows and indexes: derived, never saved
- rows: a folder's children as columns (names, ids, is, size, time), one cache shared by every
  reader: the 4N spots held once, sort 2x faster, a typed array a numeric column. a cache as
  *decided* 10-14 for paths, "yeah also we can just cache instead" (plan.org: worked out when
  read, dropped when what it read changes). a library, *mine*
- index: ids by an attribute's values (tags: each element), fed by facts of part attributes; a
  filter 36 µs not 2.3 ms; 41 B a node. a library, *mine*. it covers only what is loaded (only
  reachable exists). its readers are tracked by keys of its own (the scheduler's track and mark,
  one per value), or by places' spots if it is a node
- both need every fact of a part: places' commit and onTableChange make them; watchPath gives
  them by path only
- rows serve sorting and filtering; a row drawn on screen still reads its own node (one row reruns,
  not the list)

// rows.ts: a folder's rows as columns; reads: its names, then per row its id, is, size, date
export type Rows = { names: string[]; ids: Id[]; is: unknown[]; size: Float64Array; time: Float64Array }
export function rows(places: Places, id: Id): Rows {
	const names = places.readKeys("children", id).sort(byCodePoint), n = names.length
	const r: Rows = { names, ids: [], is: [], size: new Float64Array(n), time: new Float64Array(n) }
	for (let i = 0; i < n; i++) {
		const kid = r.ids[i] = places.readValue("children", id, [names[i]]) as Id
		r.is[i] = places.readValue("attributes", kid, ["is"])
		r.size[i] = Number(places.readValue("attributes", kid, ["size"]) ?? NaN)
		r.time[i] = places.readChange("value", kid)?.time ?? NaN
	}
	return r
}
// sorted by size: an order, the columns untouched
export const bySize = (r: Rows) => Uint32Array.from(r.size.keys()).sort((a, b) => r.size[a] - r.size[b])

// index.ts: ids by one attribute's values, from facts; a whole write or one at [name] reindexes
export function createIndex(name: string) {
	const by = new Map<unknown, Set<Id>>(), had = new Map<Id, unknown[]>()
	function put(id: Id, attributes: unknown) {
		for (const v of had.get(id) ?? []) by.get(v)?.delete(id)
		const now = [(attributes as Record<string, unknown> | undefined)?.[name] ?? []].flat()
		for (const v of now) (by.get(v) ?? by.set(v, new Set()).get(v)!).add(id)
		had.set(id, now)
	}
	const onFact = (f: Fact) => {
		if (f.part === "attributes" && f.patches.some((p) => !p.path.length || p.path[0] === name)) {
			put(f.id, f.entry?.value)
		}
	}
	return { onFact, ids: (v: unknown): ReadonlySet<Id> => by.get(v) ?? new Set() }
}

* Where it strains
- in place and the equal cut: the scheduler compares results with Object.is. a computation
  returning =readValue("children", id)= (the whole map) gets the same object after a place, so its
  readers stop there. walk (an id) and readKeys (a new list) are unaffected. choices: the whole
  read gives the live map (fast; the cut misses), or a copy (O(n), rare reads; a defensive copy,
  against "share, don't copy" 10-12), or cow stays and big folders pay per write. *open*
- code point order is an order chosen; today's three orders are accidental. a manual order stays a
  value
- rows and index are derived: per plan.org a cache is not in the graph, so they are library state,
  not nodes. visible only through what reads them (describe), not in the inspector as nodes.
  whether an index is a node: *open*
- the index is fed by facts and kept, not dropped and worked out again: a second kind of cache
  beside the decided one
- size in attributes copies a fact of the value: two writers or a computation
- part-major table: get(id) gives parts in the parts' order, not the order added
- status in attributes: a computation's attributes entry changes twice per async run, so its
  change header says "status" and keys readers of its attributes rerun; decided as is
